cmake_minimum_required(VERSION 3.28)
project(StudioHubStereoUtility VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT ON CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(STUDIOHUB_SDK_ROOT "$ENV{STUDIOHUB_SDK_ROOT}" CACHE PATH "Pinned StudioHub SDK")
set(STUDIOHUB_TOOLKIT_ROOT "$ENV{STUDIOHUB_TOOLKIT_ROOT}" CACHE PATH "StudioHub toolkit")
if(NOT EXISTS "${STUDIOHUB_SDK_ROOT}/CMakeLists.txt")
 message(FATAL_ERROR "Set STUDIOHUB_SDK_ROOT through the StudioHub toolchain profile")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${STUDIOHUB_SDK_ROOT}" sdk EXCLUDE_FROM_ALL)
smtg_add_vst3plugin(StudioHubStereoUtility src/plugin.cpp)
target_sources(StudioHubStereoUtility PRIVATE "${STUDIOHUB_SDK_ROOT}/public.sdk/source/main/linuxmain.cpp")
target_link_libraries(StudioHubStereoUtility PRIVATE sdk vstgui_support)
smtg_target_configure_version_file(StudioHubStereoUtility)
smtg_target_add_plugin_resources(StudioHubStereoUtility RESOURCES resource/editor.uidesc)
target_link_options(StudioHubStereoUtility PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
add_executable(StudioHubStereoUtilityTests src/tests.cpp)
add_executable(StudioHubStereoUtilityMeasurements src/measurements.cpp)
enable_testing()
add_test(NAME StudioHubStereoUtilityTests COMMAND StudioHubStereoUtilityTests)
add_test(NAME StudioHubStereoUtilityMeasurements COMMAND StudioHubStereoUtilityMeasurements "${CMAKE_BINARY_DIR}/stereo-utility-measurements.json")
add_custom_command(TARGET StudioHubStereoUtility POST_BUILD
 COMMAND ${Python3_EXECUTABLE} "${STUDIOHUB_TOOLKIT_ROOT}/desktop/run-isolated-validator.py" --kind steinberg --tool $<TARGET_FILE:validator> --bundle "${CMAKE_BINARY_DIR}/VST3/$<CONFIG>/StudioHubStereoUtility.vst3" --evidence "${CMAKE_BINARY_DIR}/studiohub-validation/steinberg"
 COMMAND ${Python3_EXECUTABLE} "${STUDIOHUB_TOOLKIT_ROOT}/scripts/normalize_moduleinfo.py" "${CMAKE_BINARY_DIR}/VST3/$<CONFIG>/StudioHubStereoUtility.vst3/Contents/Resources/moduleinfo.json"
 COMMAND ${Python3_EXECUTABLE} "${STUDIOHUB_TOOLKIT_ROOT}/desktop/run-isolated-validator.py" --kind metadata --tool $<TARGET_FILE:moduleinfotool> --bundle "${CMAKE_BINARY_DIR}/VST3/$<CONFIG>/StudioHubStereoUtility.vst3" --evidence "${CMAKE_BINARY_DIR}/studiohub-validation/metadata"
 VERBATIM)
add_executable(StudioHubStereoUtilityIntegrationTests src/integration_tests.cpp
 "${STUDIOHUB_SDK_ROOT}/public.sdk/source/main/linuxmain.cpp")
target_link_libraries(StudioHubStereoUtilityIntegrationTests PRIVATE sdk sdk_hosting vstgui_support)
add_test(NAME StudioHubStereoUtilityIntegrationTests COMMAND StudioHubStereoUtilityIntegrationTests)
target_sources(StudioHubStereoUtilityIntegrationTests PRIVATE
 "${STUDIOHUB_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
find_package(Threads REQUIRED)
add_executable(StudioHubStereoUtilityStateTests src/state_exchange_tests.cpp)
target_link_libraries(StudioHubStereoUtilityStateTests PRIVATE Threads::Threads)
add_test(NAME StudioHubStereoUtilityStateTests COMMAND StudioHubStereoUtilityStateTests)
